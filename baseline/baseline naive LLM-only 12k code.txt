    },
    {

    {

    {
        "id": "cash_and_equivalents_2024_12_31",
        "question": "Какова сумма денежных средств и их эквивалентов на 31 декабря 2024 года?"
    },
    {
        "id": "long_term_loans_2024_12_31",
        "question": "Какова общая сумма долгосрочных кредитов и займов на 31 декабря 2024 года?"
    },

questions_text = "\n".join(                                #формируем структуру запроса в модель
    [f"- {q['id']}: {q['question']}" for q in QUESTIONS]
)

schema_text = ",\n".join(                      #задаем структуру ответа, в ответе на каждый id вопроса будут представлены value-значение(число), currency-формат валюты, evidence found-показатель, нашла ли нейронка ответ в тексте
    [
        f'  "{q["id"]}": {{ "value": number | null, "currency": string | null, "evidence_found": boolean }}'
        for q in QUESTIONS
    ]
)


def read_pdf(path: str) -> str:                                #открываем пдф, извлекаем текст и собираем его в одну строку
    text = ""
    with pdfplumber.open(path) as pdf:
        for page in pdf.pages:
            page_text = page.extract_text()
            if page_text:
                text += page_text + "\n"
    return text


pdf_files = [                                   #читается папка с отчетами и формируется список файлов, которые будут обработаны
    f for f in os.listdir(REPORTS_DIR)
    if f.lower().endswith(".pdf")
]


with GigaChat(                                  #создаем запрос к модели для каждого отчета
    credentials="##########",
    verify_ssl_certs=False
) as chat:

    for pdf_name in pdf_files:
        print(f"\n=== REPORT: {pdf_name} ===")

        pdf_path = os.path.join(REPORTS_DIR, pdf_name)
        report_text = read_pdf(pdf_path)

        MAX_CHARS = 12000      #вводим ограничение на максимальное количество символов в отчете
        report_text = report_text[:MAX_CHARS]

        prompt = f"""                                            #прописываем промпт с задачей для нейронки
Ты — система извлечения информации из финансовой отчетности.

Ответь на вопросы СТРОГО на основе предоставленного текста.

Верни СТРОГО JSON следующего формата:
{{
{schema_text}
}}

Правила:
- Используй ТОЛЬКО предоставленный текст
- НИЧЕГО не придумывай
- Если информация не найдена — value = null и evidence_found = false

Текст отчета:
{report_text}

Вопросы:
{questions_text}
"""

        response = chat.chat(prompt)                           #получаем ответ на все вопросы по отчету
        print(response)
