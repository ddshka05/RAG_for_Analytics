questions_text = "\n".join(                                  #формируем структуру запроса 
    [f"- {q['id']}: {q['question']}" for q in QUESTIONS]
)

schema_text = ",\n".join(                                         #формируем структуру ответа
    [
        f'  "{q["id"]}": {{ "value": number | null, "currency": string | null, "evidence_found": boolean }}'
        for q in QUESTIONS
    ]
)


def read_pdf_keyword_based(path: str) -> str:              #читаем пдф, выбираем только страницы с совпадениями по ключевым словам
    collected_pages = []

    with pdfplumber.open(path) as pdf:
        for page_num, page in enumerate(pdf.pages):
            page_text = page.extract_text()
            if not page_text:
                continue

            page_text_lower = page_text.lower()

            # если на странице есть хотя бы одно ключевое слово
            if any(
                kw in page_text_lower
                for kws in KEYWORDS.values()
                for kw in kws
            ):
                collected_pages.append(
                    f"\n--- PAGE {page_num + 1} ---\n" + page_text
                )

    return "\n".join(collected_pages)


pdf_files = [                               #создаем перечень всех файлов для обработки

    if f.lower().endswith(".pdf")
]


with GigaChat(                                  #отправляем запрос модели
    credentials="#############",
    verify_ssl_certs=False
) as chat:

    for pdf_name in pdf_files:
        print(f"\n=== REPORT: {pdf_name} ===")

        pdf_path = os.path.join(REPORTS_DIR, pdf_name)
        report_text = read_pdf_keyword_based(pdf_path)

        MAX_CHARS = 60000      #вводим ограничение по символам
        report_text = report_text[:MAX_CHARS]

        prompt = f"""                      #задаем промпт для нейронки
Ты — система извлечения информации из финансовой отчетности.

Ответь на вопросы СТРОГО на основе предоставленного текста.

Верни СТРОГО JSON следующего формата:
{{
{schema_text}
}}

Правила:
- Используй ТОЛЬКО предоставленный текст
- НИЧЕГО не придумывай
- Если информация не найдена — value = null и evidence_found = false

Текст отчета:
{report_text}

Вопросы:
{questions_text}
"""

        response = chat.chat(prompt)
        print(response)                    #получаем ответ
